// Relative propagation distances for an artistic additive beam-overlap model.
function buildPrismRamp(distance=1,reverse=false){
 const spread=.055+.30*distance,width=.21,centers=[.5+spread,.5,.5-spread];
 return Array.from({length:8},(_,i)=>{const t=i/7,x=reverse?1-t:t,rgb=centers.map(center=>Math.exp(-.5*((x-center)/width)**2)),hex='#'+rgb.map(v=>Math.round(encodeSRGB(v)*255).toString(16).padStart(2,'0')).join('');return {t,hex};});
}

 const keys=buildPrismRamp(distance);gradientPresets.push({id:'prism-'+id,name:'Prism · '+name,category:'Optics',description:'Artistic linear-light RGB beam overlap at relative distance '+distance+'. Fixed beam width; increased separation. Not calibrated prism optics or a spatial aberration shader.',colors:keys.map(k=>k.hex),positions:keys.map(k=>k.t)});
}
