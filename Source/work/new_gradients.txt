// Native OKLab coordinates: L in percent, chroma scaled by 250.
function perceptualRampColor(l,c,h){const angle=h*Math.PI/180,fit=fitLabToSRGB(l,c*Math.cos(angle),c*Math.sin(angle));return labColor(fit.l,fit.a,fit.b).hex;}
function buildPerceptualRamp({family='sequential',hue=250,span=70,chroma=45,count=8,layer='emission',phase=0}){
 const ceiling=layer==='smoke'?84:97,floor=layer==='smoke'?12:7;
 return Array.from({length:count},(_,i)=>{const t=i/(count-1);let l,c,h;
 if(family==='diverging'){const d=Math.abs(2*t-1);l=95-61*d;c=chroma*d**.8;h=t<.5?hue:hue+span;}
 else if(family==='cyclic'){l=68+5*Math.sin(2*Math.PI*t);c=chroma*.78;h=hue+360*t;}
 else if(family==='accent'){l=floor+(ceiling-floor)*t;c=chroma*Math.sin(Math.PI*t)**.65;h=hue+(t<.65?span*t*.35:span+100*(t-.65));}
 else{l=floor+(ceiling-floor)*t;c=chroma*Math.sin(Math.PI*t)**.75;h=hue+span*(t-.5)+phase*Math.sin(Math.PI*t);}
 return {t,hex:perceptualRampColor(l,c,h)};
 });
}
